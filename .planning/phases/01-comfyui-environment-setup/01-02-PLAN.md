```bash

  --local-dir-use-symlinks False

# SyncNet model (1.61 GB)
huggingface-cli download ByteDance/LatentSync-1.6 stable_syncnet.pt \
  --local-dir /data/workspace/comfyui-wan/custom_nodes/ComfyUI-LatentSyncWrapper/checkpoints/ \
  --local-dir-use-symlinks False

# Whisper tiny model (~75 MB)
huggingface-cli download ByteDance/LatentSync-1.6 whisper/tiny.pt \
  --local-dir /data/workspace/comfyui-wan/custom_nodes/ComfyUI-LatentSyncWrapper/checkpoints/whisper/ \
  --local-dir-use-symlinks False

# VAE model (sd-vae-ft-mse, ~335 MB)
huggingface-cli download stabilityai/sd-vae-ft-mse diffusion_pytorch_model.safetensors config.json \
  --local-dir /data/workspace/comfyui-wan/custom_nodes/ComfyUI-LatentSyncWrapper/checkpoints/vae/ \
  --local-dir-use-symlinks False

# Auxiliary models (face detection, small)
```
