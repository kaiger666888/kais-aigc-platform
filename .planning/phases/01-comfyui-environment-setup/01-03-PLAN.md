  'IPAdapter': 'IP-Adapter Plus face conditioning',
  'InstantID': 'InstantID zero-shot face preservation',
  'PhotoMaker': 'PhotoMaker Plus multi-reference character',
  'RIFE': 'RIFE frame interpolation'
}
all_ok = True
for keyword, desc in targets.items():
    matching = [k for k in n.keys() if keyword.lower() in k.lower()]
    if matching:
        print(f'OK: {keyword} -> {matching}')
    else:
        print(f'FAIL: {keyword} NOT FOUND ({desc})')
        all_ok = False
if not all_ok:
    sys.exit(1)
print('All nodes registered successfully')
"
