# Task: Add multi-select + batch "Set as Primary View" to Asset Library

## Context
The infinite-canvas asset management page (`AssetLibrary.tsx`) currently shows character/scene variant cards.

## Backend (already exists — no changes needed)

Add this near line 297 (after `searchAssets`):

```typescript
/**
 * Update a single asset's metadata via PATCH /api/v1/assets/:id.
 * Used for isPrimaryView, tags, state, etc.
 */
export async function updateAsset(
  assetId: number,
  updates: { isPrimaryView?: boolean; tags?: string; state?: string; name?: string; describe?: string },
): Promise<void> {
  const res = await fetch(`${API_BASE}/v1/assets-registry/${assetId}`, {
    method: 'PATCH',
    headers: { 'Content-Type': 'application/json' },
    body: JSON.stringify(updates),
  })
  if (!res.ok) throw new ApiError(`HTTP ${res.status}`, 'network', res.status)
}
```

```typescript
const toggleSelect = (id: number) => {
  setSelected(prev => {
    const next = new Set(prev)
    if (next.has(id)) next.delete(id)
    else next.add(id)
    return next
  })
}

const selectAll = () => {
  setSelected(new Set(items.map(a => a.id!).filter(Boolean)))
}

const clearSelection = () => {
  setSelected(new Set())
}

const handleBatchPrimary = async () => {
  if (selected.size === 0) return
  setBatchLoading(true)
  try {
    const ids = Array.from(selected)
    // Set selected as primary, unset others of same type
    // First, get all items' types from selected
    const selectedTypes = new Set(items.filter(a => selected.has(a.id!)).map(a => a.type))
    
    // For each selected type, find all items of that type in the full asset list and unset their primary
    const sameTypeItems = assets.filter(d => {
      const item = assetDetailToItem(d)
      return selectedTypes.has(item.type)
    })
    
    // Unset all same-type items first
    for (const d of sameTypeItems) {
      if (d.isPrimaryView && !selected.has(d.id)) {
        await updateAsset(d.id, { isPrimaryView: false })
      }
    }
    
    // Set selected items as primary
    for (const id of ids) {
      await updateAsset(id, { isPrimaryView: true })
    }
    
    showToast(`已将 ${ids.length} 个资产设为正式使用版本`, 'success')
    setSelectMode(false)
    setSelected(new Set())
    await reload()
  } catch (err) {
    showToast('批量设置失败: ' + (err as Error).message, 'error')
  } finally {
    setBatchLoading(false)
  }
}
```
